import re

def parse_duration(s: str) -> int:
    """Return the number of seconds in a duration string like '1h30m', '45s' or '2h'.

    Format: optional hours, minutes and seconds in h-m-s order; case and
    whitespace are ignored. Raises ValueError for empty input, a bare number,
    an unknown unit or parts out of order.
    """
    t = re.sub(r"\s+", "", s).lower()
    m = re.fullmatch(r"(?:(\d+)h)?(?:(\d+)m)?(?:(\d+)s)?", t)
    if not t or m is None:
        raise ValueError(f"invalid duration: {s!r}")
    h, mi, sec = (int(x or 0) for x in m.groups())
    return h * 3600 + mi * 60 + sec

import re

def parse_duration(s: str) -> int:
    """Convert a duration string such as '1h30m', '45s' or '2h' to seconds.

    Optional h, m and s parts must appear in that order; whitespace and case
    are ignored. Raises ValueError for an empty string, a number without a
    unit, an unknown unit or parts in the wrong order.

    t = "".join(s.split()).lower()
    m = re.fullmatch(r"(?:(\d+)h)?(?:(\d+)m)?(?:(\d+)s)?", t)
    if not t or not m:
        raise ValueError(f"invalid duration: {s!r}")
    h, mi, sec = (int(g or 0) for g in m.groups())
    return h * 3600 + mi * 60 + sec
