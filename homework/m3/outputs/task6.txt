import re

ENVS = {"dev": "dev", "stg": "stg", "prd": "prd", "development": "dev", "staging": "stg", "production": "prd"}


def tf_name(project: str, env: str, component: str, index: int | None = None) -> str:
    e = ENVS.get(env.strip().lower())
    if e is None:
        raise ValueError(f"unknown env: {env!r}")

    def clean(part: str) -> str:
        c = re.sub(r"[^a-z0-9]+", "-", part.lower()).strip("-")
        if not c:
            raise ValueError(f"empty name part: {part!r}")
        return c

    p, c = clean(project), clean(component)
    suffix = ""
    if index is not None:
        if not 1 <= index <= 99:
            raise ValueError(f"index out of range: {index}")
        suffix = f"-{index:02d}"
    room = 32 - len(f"{p}-{e}-") - len(suffix)
    if room < 1:
        raise ValueError("name cannot fit in 32 characters")
    if len(c) > room:
        cut = c.rfind("-", 0, room + 1)
        c = c[:cut] if cut > 0 else c[:room]
    return f"{p}-{e}-{c}{suffix}"

import re

ENVS = {"dev": "dev", "stg": "stg", "prd": "prd", "development": "dev", "staging": "stg", "production": "prd"}


def tf_name(project: str, env: str, component: str, index: int | None = None) -> str:
    """Build a cloud resource name: <project>-<env>-<component>[-<NN>], all lowercase.

    env is dev, stg or prd (aliases development, staging, production). In project and
    component every run of characters outside [a-z0-9] becomes one hyphen, trimmed at
    both ends. index is 1..99, written with two digits. Names are at most 32 characters:
    only the component is shortened, at the last hyphen that fits, else hard cut.

    Raises:
        ValueError: unknown env, empty project or component after cleaning,
            index outside 1..99, or a name that cannot fit in 32 characters.
    """
    e = ENVS.get(env.strip().lower())
    if e is None:
        raise ValueError(f"unknown env: {env!r}")

    def clean(part: str) -> str:
        c = re.sub(r"[^a-z0-9]+", "-", part.lower()).strip("-")
        if not c:
            raise ValueError(f"empty name part: {part!r}")
        return c

    p, c = clean(project), clean(component)
    suffix = ""
    if index is not None:
        if not 1 <= index <= 99:
            raise ValueError(f"index out of range: {index}")
        suffix = f"-{index:02d}"
    room = 32 - len(f"{p}-{e}-") - len(suffix)
    if room < 1:
        raise ValueError("name cannot fit in 32 characters")
    if len(c) > room:
        cut = c.rfind("-", 0, room + 1)
        c = c[:cut] if cut > 0 else c[:room]
    return f"{p}-{e}-{c}{suffix}"

import re
import unicodedata


def slugify(title: str, max_len: int = 40) -> str:
    ascii_title = unicodedata.normalize("NFKD", title).encode("ascii", "ignore").decode()
    slug = re.sub(r"[^a-z0-9]+", "-", ascii_title.lower()).strip("-")
    if len(slug) <= max_len:
        return slug
    cut = slug.rfind("-", 0, max_len + 1)
    return slug[:cut] if cut > 0 else slug[:max_len]
