import re

def parse_duration(s: str) -> int:
    """Convert a duration such as '1h30m', '45s' or '2h' into seconds.

    Accepts optional hour, minute and second parts in that order (h, m, s);
    whitespace and letter case are ignored, and values may exceed 59.

    Raises:
        ValueError: if the string is empty, has a number without a unit,
            uses an unknown unit or has the parts in the wrong order.
    """
    t = "".join(s.split()).lower()
    m = re.fullmatch(r"(?:(\d+)h)?(?:(\d+)m)?(?:(\d+)s)?", t)
    if not t or not m:
        raise ValueError(f"invalid duration: {s!r}")
    h, mi, sec = (int(g) if g else 0 for g in m.groups())
    return h * 3600 + mi * 60 + sec
