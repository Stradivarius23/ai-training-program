def parse_duration(s: str) -> int:
    units = {"h": 3600, "m": 60, "s": 1}
    total, num, seen = 0, "", False
    for ch in s.lower():
        if ch.isspace():
            continue
        if ch.isdigit():
            num += ch
        elif ch in units and num:
            total += int(num) * units[ch]
            num, seen = "", True
        else:
            raise ValueError(f"invalid duration: {s!r}")
    if num or not seen:
        raise ValueError(f"invalid duration: {s!r}")
    return total
